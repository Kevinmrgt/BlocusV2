### API Service Pattern
[Source: docs/architecture/frontend-architecture.md#service-layer]

```typescript
// src/services/api/favorites.ts
import { supabase } from '@/lib/supabase';
import type { BoulderWithDetails } from '@/types/models';

export async function getFavorites(userId: string): Promise<BoulderWithDetails[]> {
  const { data, error } = await supabase
    .from('favorites')
    .select(`
      id,
      created_at,
      boulder:boulders (
        id,
        title,
        description,
        difficulty,
        wall_id,
        boulder_photos (id, url, order_index)
      )
    `)
    .eq('user_id', userId)
    .order('created_at', { ascending: false });

  if (error) throw error;
  return data.map(f => f.boulder);
}

export async function addFavorite(userId: string, boulderId: string): Promise<void> {
  const { error } = await supabase
    .from('favorites')
    .insert({ user_id: userId, boulder_id: boulderId });

  if (error) throw error;
}

export async function removeFavorite(userId: string, boulderId: string): Promise<void> {
  const { error } = await supabase
    .from('favorites')
    .delete()
    .eq('user_id', userId)
    .eq('boulder_id', boulderId);

  if (error) throw error;
}
```

### React Query Hook Pattern
[Source: docs/architecture/frontend-architecture.md#custom-hooks]

```typescript
// src/hooks/useFavorites.ts
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { getFavorites, addFavorite, removeFavorite } from '@/services/api/favorites';
import { useAuth } from '@/providers/AuthProvider';

export function useFavorites() {
  const { user } = useAuth();

  return useQuery({
    queryKey: ['user', 'favorites', user?.id],
    queryFn: () => getFavorites(user!.id),
    enabled: !!user?.id,
  });
}

export function useToggleFavorite() {
  const queryClient = useQueryClient();
  const { user } = useAuth();

  return useMutation({
    mutationFn: async ({ boulderId, isFavorited }: { boulderId: string; isFavorited: boolean }) => {
      if (isFavorited) {
        await removeFavorite(user!.id, boulderId);
      } else {
        await addFavorite(user!.id, boulderId);
      }
    },
    onSuccess: () => {
      // Invalidate favorites list
      queryClient.invalidateQueries({ queryKey: ['user', 'favorites', user!.id] });
      // Invalidate user profile (favorites_count)
      queryClient.invalidateQueries({ queryKey: ['user', 'profile', user!.id] });
      // Invalidate boulder detail (is_favorited)
      queryClient.invalidateQueries({ queryKey: ['boulder'] });
    },
  });
}
```

### RLS Policies for Favorites
[Source: docs/architecture/backend-architecture-row-level-security.md#favorites-table]

```sql
-- Users can only see their own favorites
CREATE POLICY "favorites_select_own"
  ON favorites FOR SELECT
  USING (auth.uid() = user_id);

-- Users can add their own favorites
CREATE POLICY "favorites_insert_own"
  ON favorites FOR INSERT
  WITH CHECK (auth.uid() = user_id);

-- Users can remove their own favorites
CREATE POLICY "favorites_delete_own"
  ON favorites FOR DELETE
  USING (auth.uid() = user_id);
```
